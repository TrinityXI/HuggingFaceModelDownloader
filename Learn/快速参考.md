# Go 语言快速参考

> 常用语法和模式速查表

## 📌 基础语法

### 变量声明

```go
var x int                    // 零值初始化
var y = 10                  // 类型推断
z := 20                     // 短变量声明（仅函数内）
const PI = 3.14             // 常量

// 批量声明
var (
	name string
	age  int
)
```

### 基本类型

```go
bool                        // true, false
string                      // "hello"
int, int8, int16, int32, int64
uint, uint8, uint16, uint32, uint64
byte                        // uint8 别名
rune                        // int32 别名，表示 Unicode 码点
float32, float64
complex64, complex128
```

### 函数

```go
// 基本函数
func add(a, b int) int {
	return a + b
}

// 多返回值
func swap(x, y string) (string, string) {
	return y, x
}

// 命名返回值
func divide(a, b int) (result int, err error) {
	if b == 0 {
		err = errors.New("division by zero")
		return
	}
	result = a / b
	return
}

// 可变参数
func sum(nums ...int) int {
	total := 0
	for _, n := range nums {
		total += n
	}
	return total
}
```

## 📦 控制结构

### if 语句

```go
// 基本形式
if x > 0 {
	// ...
}

// 带初始化
if v := getValue(); v > 0 {
	// v 仅在此作用域可用
}

// if-else
if x > 0 {
	// ...
} else if x < 0 {
	// ...
} else {
	// ...
}
```

### for 循环

```go
// 传统 for
for i := 0; i < 10; i++ {
	fmt.Println(i)
}

// while 风格
for condition {
	// ...
}

// 无限循环
for {
	// ...
	if done {
		break
	}
}

// range 遍历
for i, v := range slice {
	fmt.Println(i, v)
}

for key, value := range myMap {
	fmt.Println(key, value)
}

// 忽略索引/键
for _, v := range slice {
	fmt.Println(v)
}
```

### switch 语句

```go
// 基本 switch
switch value {
case 1:
	// ...
case 2, 3:
	// ...
default:
	// ...
}

// 无表达式 switch
switch {
case x > 0:
	// ...
case x < 0:
	// ...
default:
	// ...
}

// 类型 switch
switch v := interface{}(x).(type) {
case int:
	// v 是 int
case string:
	// v 是 string
default:
	// ...
}
```

### defer

```go
func example() {
	defer fmt.Println("world")  // 最后执行
	fmt.Println("hello")
	// 输出: hello world
}

// 常用于资源清理
file, err := os.Open("file.txt")
if err != nil {
	return err
}
defer file.Close()  // 确保关闭
```

## 🏗️ 数据结构

### 数组


## 🎯 结构体

### 定义和使用

```go
// 定义
type Person struct {
	Name string
	Age  int
}

// 创建
p1 := Person{"Alice", 30}
p2 := Person{Name: "Bob", Age: 25}
p3 := Person{Name: "Charlie"}  // Age = 0
```

### 方法

```go
// 值接收者
func (p Person) GetName() string {
	return p.Name
}

// 指针接收者
func (p *Person) SetAge(age int) {
	p.Age = age
}

// 使用
person := Person{"Alice", 30}
fmt.Println(person.GetName())
person.SetAge(31)
```

## 🔗 接口

### 定义和实现

```go

type Reader interface {
	Read(p []byte) (n int, err error)
}

// 隐式实现
type MyReader struct{}

func (r MyReader) Read(p []byte) (n int, err error) {
	// 实现

// MyReader 自动实现了 Reader 接口
```

### 常用接口

```go
// 空接口（可以接受任何类型）
var x interface{}
x = 42
x = "hello"

// 类型断言
value, ok := x.(string)
if ok {
	fmt.Println("x is a string:", value)
}

// 类型 switch
switch v := x.(type) {

	fmt.Println("int:", v)

	fmt.Println("unknown type")

```

## 🚀 并发

### Goroutine

```go
// 启动 goroutine
go func() {
	fmt.Println("Hello from goroutine")
}()

// 带参数
go processItem(item)
```

### Channel

```go
// 创建
ch := make(chan int)           // 无缓冲
ch := make(chan int, 10)       // 有缓冲，容量 10

// 发送和接收
ch <- 42                        // 发送
value := <-ch                   // 接收
value, ok := <-ch               // 检查 channel 是否关闭

// 关闭
close(ch)

// 遍历
for v := range ch {
	fmt.Println(v)
}
```

### Select

```go
select {
case v := <-ch1:
	// 从 ch1 接收
case ch2 <- v:
	// 发送到 ch2
case <-time.After(time.Second):
	// 超时
default:
	// 所有 case 都不可用时执行
}
```

### Sync 包

```go
// Mutex
var mu sync.Mutex
mu.Lock()
// 临界区
mu.Unlock()

// RWMutex
var rw sync.RWMutex
rw.RLock()   // 读锁
// 读操作
rw.RUnlock()

rw.Lock()    // 写锁
// 写操作
rw.Unlock()

// WaitGroup
var wg sync.WaitGroup
for i := 0; i < 10; i++ {
	wg.Add(1)
	go func(id int) {
		defer wg.Done()
		// 工作
	}(i)
}
wg.Wait()  // 等待所有完成

// Once
var once sync.Once
once.Do(func() {
	// 只执行一次
})
```

## ⚠️ 错误处理

### 创建错误

```go
// errors.New
err := errors.New("something went wrong")

// fmt.Errorf
err := fmt.Errorf("failed to open %s", filename)

// 包装错误
err := fmt.Errorf("outer error: %w", innerErr)
```

### 检查错误

```go
// 基本检查
if err != nil {
	return err
}

// errors.Is
if errors.Is(err, ErrNotFound) {
	// 特定错误
}

// errors.As
var netErr *NetworkError
if errors.As(err, &netErr) {
	// 类型断言
	fmt.Println(netErr.Code)
}
```

### 选项模式

```go
type Option func(*Config)

func WithTimeout(d time.Duration) Option {
	return func(c *Config) {
		c.Timeout = d
	}
}

func NewClient(opts ...Option) *Client {
	cfg := &Config{}
	for _, opt := range opts {
		opt(cfg)
	}
	return &Client{config: cfg}
}

// 使用
client := NewClient(
	WithTimeout(5*time.Second),
	WithRetries(3),
)
```

### 单例

```go
var (
	instance *Singleton
	once     sync.Once
)

func GetInstance() *Singleton {
	once.Do(func() {
		instance = &Singleton{}
	})
	return instance
}
```
